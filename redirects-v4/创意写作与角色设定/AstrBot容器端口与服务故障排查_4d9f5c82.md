```yaml
# 陛下专属的小说生成器云端炼丹炉架构
version: '3.8'

services:
  novel-writer-api:
    # 这里可以替换成您在 Github 找到的具体开源小说生成器镜像

    container_name: auto_novel_generator
    restart: always # 防御性编程：崩溃后赛博重生
    working_dir: /app
    volumes:
      # 把服务器宿主机的 data 目录挂载进去，保证您的小说存档永不丢失！

      - ./script:/app/script

      resources:
        limits:
          memory: 300M
    # 保持容器后台长存，方便您随时用手机 SSH 进去跑脚本
    command: tail -f /dev/null 

```
