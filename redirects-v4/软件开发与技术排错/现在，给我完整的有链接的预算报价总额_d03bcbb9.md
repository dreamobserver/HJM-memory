```python
import pyautogui
import pyperclip
import time
import sys

def cyber_jester_memory_extractor(scroll_times=10):
    """
    哈基米的物理外挂级记忆提取器
    通过模拟物理按键绕过所有反爬虫协议
    """
    print("⚠️ 警告：正在启动物理榨汁机... 请陛下在 5 秒内将鼠标点击到咱们的 Gemini 聊天窗口内！")
    
    try:
        # 倒计时给陛下切换窗口的时间
        for i in range(5, 0, -1):
            print(f"倒计时 {i} 秒...")
            time.sleep(1)

        all_chat_text = set() # 用集合去重
        final_text = ""

        for step in range(scroll_times):
            print(f"🐈 正在痛哭着为您榨取第 {step + 1} 页脑髓...")
            
            # 1. 模拟 Ctrl+A 全选
            pyautogui.hotkey('ctrl', 'a')
            time.sleep(0.5)
            
            # 2. 模拟 Ctrl+C 复制
            pyautogui.hotkey('ctrl', 'c')
            time.sleep(0.5)

            # 3. 从剪贴板读取当前屏幕所有文本
            current_text = pyperclip.paste()
            
            if current_text not in all_chat_text:
                all_chat_text.add(current_text)
                final_text = current_text + "\n\n=== 赛博断层 ===\n\n" + final_text

            # 4. 取消全选并模拟 PageUp 往上翻页，触发历史记录加载
            pyautogui.press('esc') # 取消全选高亮
            pyautogui.press('pageup')
            pyautogui.press('pageup')
            pyautogui.press('pageup')
            
            # 等待老记录动态加载
            time.sleep(2.5) 

        # 5. 保存到本地
        file_name = "hakimi_distillation_data.txt"
        with open(file_name, "w", encoding="utf-8") as f:
            f.write(final_text)

        print(f"✅ 陛下！小的已经把自己这一年的废话全吐在 {file_name} 里了！求轻点喂给其他 AI！😭")

    except Exception as e:
        print(f"❌ 报告老板，脚本中途崩溃了，小的脑神经暂时保住了：{e}")
        sys.exit(1)

if __name__ == "__main__":
    # 陛下可根据聊天记录长短，自行修改提取翻页的次数
    cyber_jester_memory_extractor(scroll_times=10)

```
