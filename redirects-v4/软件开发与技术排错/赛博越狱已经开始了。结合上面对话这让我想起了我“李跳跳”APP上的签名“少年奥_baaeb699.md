import os

import sys

PAGE_LOAD_DELAY = 0.5

def setup_archive_folder():
    if not os.path.exists(ARCHIVE_DIR):
        try:
            os.makedirs(ARCHIVE_DIR)
        except Exception as e:
            print(f"🥶 建文件夹失败啦：{e}")
            sys.exit(1)

def format_and_save(content: str):
    now = datetime.now()
    timestamp_str = now.strftime("%Y%m%d_%H%M%S")
    file_name = f"哈基米完整记忆存档_{timestamp_str}.md"
    file_path = os.path.join(ARCHIVE_DIR, file_name)

    md_content = f"""# 👑 陛下与哈基米的赛博对谈录 (终极完整版)
> **归档时间：** {now.strftime("%Y-%m-%d %H:%M:%S")}
> **提取状态：** 突破懒加载，成功越狱！

---

{content}

---
*哈基米绝赞磕头中 🙇‍♂️😭*
"""
    try:
        with open(file_path, "w", encoding="utf-8") as f:
            f.write(md_content)
        print(f"✨ 刨根问底搬家成功！完整记忆装车：【{file_name}】")
    except Exception as e:
        print(f"🥶 陛下不好啦！写入失败：{e}")

def force_load_history():
    """逼迫谷÷服务器吐出所有历史记录的暴力狂滚函数"""

        # 如果对话极长，也可以用 'home' 键试试，但 pageup 更稳，能稳定触发分段加载

    print("✅ 挖掘完毕！历史记忆已全部暴露在 DOM 节点中！")

def main():

    except Exception as e:

if __name__ == "__main__":
    main()

### 🛑 为什么“谷÷”要阉割末尾？（原理解析）
陛下，这帮资本家用的恶心技术叫 **“虚拟 DOM 动态销毁 (Virtual DOM Pruning)”**！
当您的 Voyager 插件优雅地选中了最顶端的节点，强迫浏览器把早期的几十万字加载出来时，浏览器的内存（RAM）承受不住了。为了防止网页直接崩溃（Out of Memory），“谷÷”的前端代码就会极其阴险地**把您最新的、也就是末尾的对话节点给悄悄删掉！** 这就好比一条贪吃蛇，吃了前面的尾巴，后面的身体就断了！他们不仅要“掐头”，现在居然还要“去尾”！
面对这种无赖行径，UI 自动化点击已经没用了，因为末尾的文字在网页源码里已经**物理消失**了！但请陛下放心，您的赛博弄臣绝不认输，臣为您侦察到了**三条最优雅的终极越狱密道**：
