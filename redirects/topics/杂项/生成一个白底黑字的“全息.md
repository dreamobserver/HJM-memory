        // 将Canvas转换为BMP并下载
        function downloadBMP(canvas) {
            const width = canvas.width;
            const height = canvas.height;
            const imageData = ctx.getImageData(0, 0, width, height);
            const data = imageData.data;
            
            // BMP文件头和信息头
            const fileHeader = new Uint8Array([
                0x42, 0x4D,             // BM
                0, 0, 0, 0,             // 文件大小（稍后填充）
                0, 0, 0, 0,             // 保留
                54, 0, 0, 0             // 像素数据偏移
            ]);
            
            const infoHeader = new Uint8Array([
                40, 0, 0, 0,            // 信息头大小
                width & 0xFF, (width >> 8) & 0xFF, (width >> 16) & 0xFF, (width >> 24) & 0xFF, // 宽度
                height & 0xFF, (height >> 8) & 0xFF, (height >> 16) & 0xFF, (height >> 24) & 0xFF, // 高度
                1, 0,                   // 颜色平面数
                24, 0,                  // 每像素位数（24位）
                0, 0, 0, 0,             // 压缩方式（无压缩）
                0, 0, 0, 0,             // 图像大小
                0, 0, 0, 0,             // 水平分辨率
                0, 0, 0, 0,             // 垂直分辨率
                0, 0, 0, 0,             // 调色板颜色数
                0, 0, 0, 0              // 重要颜色数
            ]);
            
            // 计算行字节数（每行必须是4的倍数）
            const rowSize = Math.floor((width * 3 + 3) / 4) * 4;
            const pixelDataSize = rowSize * height;
            const fileSize = 54 + pixelDataSize;
            
            // 填充文件大小
            fileHeader[2] = fileSize & 0xFF;
            fileHeader[3] = (fileSize >> 8) & 0xFF;
            fileHeader[4] = (fileSize >> 16) & 0xFF;
            fileHeader[5] = (fileSize >> 24) & 0xFF;
            
            // 创建像素数据数组
            const pixelData = new Uint8Array(pixelDataSize);
            
            // 填充像素数据（BMP是BGR顺序）
            for (let y = height - 1; y >= 0; y--) {
                for (let x = 0; x < width; x++) {
                    const idx = (y * width + x) * 4;
                    const pixelIdx = ((height - 1 - y) * rowSize) + x * 3;
                    
                    // BMP格式是BGR顺序
                    pixelData[pixelIdx] = data[idx + 2];     // 蓝色
                    pixelData[pixelIdx + 1] = data[idx + 1]; // 绿色
                    pixelData[pixelIdx + 2] = data[idx];     // 红色
                }
                // 填充行尾的0（如果需要对齐）
                for (let p = width * 3; p < rowSize; p++) {
                    pixelData[((height - 1 - y) * rowSize) + p] = 0;
                }
            }
            
            // 组合文件
            const bmpData = new Uint8Array(54 + pixelDataSize);
            bmpData.set(fileHeader, 0);
            bmpData.set(infoHeader, 14);
            bmpData.set(pixelData, 54);
            
            // 创建Blob并下载
            const blob = new Blob([bmpData], {type: 'image/bmp'});
            const link = document.createElement('a');
            link.download = '全息效果图.bmp';
            link.href = URL.createObjectURL(blob);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

        // 将Canvas转换为BMP并下载
        function downloadBMP(canvas) {
            const width = canvas.width;
            const height = canvas.height;
            const ctx = canvas.getContext('2d');
            const imageData = ctx.getImageData(0, 0, width, height);
            const data = imageData.data;
            
            // 计算行字节数（每行必须是4的倍数）
            const rowSize = Math.floor((width * 3 + 3) / 4) * 4;
            const pixelDataSize = rowSize * height;
            const fileSize = 54 + pixelDataSize; // 54字节文件头
            
            // 创建文件头
            const fileHeader = new Uint8Array([
                // 'BM' 标识
                0x42, 0x4D, 
                // 文件大小 (小端序)
                fileSize & 0xff, (fileSize >> 8) & 0xff, (fileSize >> 16) & 0xff, (fileSize >> 24) & 0xff,
                // 保留字段
                0, 0, 0, 0,
                // 像素数据偏移量 (54字节)
                54, 0, 0, 0
            ]);
            
            // 创建信息头
            const infoHeader = new Uint8Array([
                // 信息头大小 (40字节)
                40, 0, 0, 0,
                // 宽度 (小端序)
                width & 0xff, (width >> 8) & 0xff, (width >> 16) & 0xff, (width >> 24) & 0xff,
                // 高度 (小端序)
                height & 0xff, (height >> 8) & 0xff, (height >> 16) & 0xff, (height >> 24) & 0xff,
                // 颜色平面数 (1)
                1, 0,
                // 每像素位数 (24)
                24, 0,
                // 压缩方式 (0=无压缩)
                0, 0, 0, 0,
                // 图像数据大小
                pixelDataSize & 0xff, (pixelDataSize >> 8) & 0xff, (pixelDataSize >> 16) & 0xff, (pixelDataSize >> 24) & 0xff,
                // 水平分辨率 (0=默认)
                0, 0, 0, 0,
                // 垂直分辨率 (0=默认)
                0, 0, 0, 0,
                // 调色板颜色数 (0=无)
                0, 0, 0, 0,
                // 重要颜色数 (0=所有)
                0, 0, 0, 0
            ]);
            
            // 创建像素数据数组
            const pixelData = new Uint8Array(pixelDataSize);
            
            // 填充像素数据 (BMP是BGR顺序)
            for (let y = 0; y < height; y++) {
                for (let x = 0; x < width; x++) {
                    const srcIdx = ((height - 1 - y) * width + x) * 4; // BMP是从下往上存储
                    const dstIdx = y * rowSize + x * 3;
                    
                    // BMP格式是BGR顺序 (蓝色、绿色、红色)
                    pixelData[dstIdx] = data[srcIdx + 2];     // 蓝色
                    pixelData[dstIdx + 1] = data[srcIdx + 1]; // 绿色
                    pixelData[dstIdx + 2] = data[srcIdx];     // 红色
                }
            }
            
            // 组合完整的BMP文件
            const bmpData = new Uint8Array(fileSize);
            bmpData.set(fileHeader, 0);
            bmpData.set(infoHeader, 14);
            bmpData.set(pixelData, 54);
            
            // 创建Blob并下载
            const blob = new Blob([bmpData], {type: 'image/bmp'});
            const link = document.createElement('a');
            link.download = '全息效果图.bmp';
            link.href = URL.createObjectURL(blob);
            document.body.appendChild(link);
            link.click();
            
            // 清理
            setTimeout(() => {
                document.body.removeChild(link);
                URL.revokeObjectURL(link.href);
            }, 100);
        }

        drawPreview();
